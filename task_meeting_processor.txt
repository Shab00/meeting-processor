from app.config import settings

class Base(DeclarativeBase):
    pass

engine = create_engine(settings.database_url, pool_pre_ping=True)
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False)

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

class ActionItem(BaseModel):
    owner: str

    due_date: str | None = None

class Decision(BaseModel):

class Risk(BaseModel):
    description: str

    source_quote: str

class MeetingAnalysis(BaseModel):
    summary: str
    action_items: list[ActionItem]
    decisions: list[Decision]
    risks: list[Risk]

class AnalyseRequest(BaseModel):
    title: str
    transcript: str

class AnalyseResponse(BaseModel):
    meeting_id: UUID
    analysis: MeetingAnalysis

class MeetingOut(BaseModel):
    id: UUID
    title: str
    created_at: datetime
    analysis: MeetingAnalysis | None = None
    model_config = ConfigDict(from_attributes=True)

from sqlalchemy import Column, String, DateTime, JSON

Use the OpenAI Agents SDK:

from agents import Agent, Runner, handoff
from app.config import settings
from app.agents.action_items import action_items_agent
from app.agents.decisions import decisions_agent
from app.agents.risks import risks_agent
from app.agents.summary import summary_agent

triage_agent = Agent(
    name="Triage",
    instructions=(
        "You receive a meeting transcript. Identify which specialists are "
        "needed and hand off to them. Always hand off to summary, then "
        "to action_items, then decisions, then risks, in that order."
    ),
    model=settings.agent_model,
    handoffs=[
        handoff(action_items_agent),
        handoff(decisions_agent),
        handoff(risks_agent),
        handoff(summary_agent),
    ],
)

async def run_triage(transcript: str):
    result = await Runner.run(triage_agent, input=transcript)
    return result

Check the installed openai-agents SDK for the correct API shape
(v0.x may use Runner.run_sync vs Runner.run). If the SDK version installed
uses a different signature, adapt — but keep the handoff structure.

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from slowapi import Limiter, _rate_limit_exceeded_handler
from slowapi.util import get_remote_address
from slowapi.errors import RateLimitExceeded

from app.config import settings
from app.routes import analyse, meetings

limiter = Limiter(key_func=get_remote_address)
app = FastAPI(title="Meeting Processor", version="0.1.0")
app.state.limiter = limiter
app.add_exception_handler(RateLimitExceeded, _rate_limit_exceeded_handler)

app.add_middleware(
    CORSMiddleware,
    allow_origins=[settings.allowed_origin],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(analyse.router)
app.include_router(meetings.router)

@app.get("/health")
def health():
    return {"status": "ok"}
